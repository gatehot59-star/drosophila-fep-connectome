def capture(command: list[str], timeout: float = 20.0) -> dict:
    """Record raw output and stop the whole child process group on timeout."""
    started = now()
    clock = time.monotonic()

        command, stdout=subprocess.PIPE, stderr=subprocess.PIPE,

        stdout, stderr = process.communicate(timeout=timeout)

        "argv": command, "started_utc": started, "ended_utc": now(),
        "wall_seconds": time.monotonic() - clock,
